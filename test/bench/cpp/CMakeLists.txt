cmake_minimum_required(VERSION 3.20)
project(capnp_es_rpc_bench CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(CapnProto CONFIG REQUIRED)
get_filename_component(CAPNPC_SRC_PREFIX ../data/sandstorm ABSOLUTE)
capnp_generate_cpp(CAPNP_SRCS CAPNP_HDRS ${CAPNPC_SRC_PREFIX}/workloads.capnp)
add_executable(rpc_bench_server server.cxx ${CAPNP_SRCS})
target_include_directories(rpc_bench_server PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(rpc_bench_server PRIVATE CapnProto::capnp-rpc)
